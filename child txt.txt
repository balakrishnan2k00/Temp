def update_cover_page(self,excel_file):
    
    try:
        wb = load_workbook(excel_file, keep_vba=True)
        # Check if 'Cover Page' exists in the workbook
        if 'Cover Page' not in wb.sheetnames:
            raise Exception("The worksheet 'Cover Page' does not exist in the workbook.")
        
        cover_sheet = wb['Cover Page']
        
        cover_sheet['E5'].value =  self.Report_Name
        cover_sheet['E7'].value = self.Reporting_Period
        cover_sheet['E9'].value = self.Data_Process
        cover_sheet['E11'].value = self.Data_File_Name
        cover_sheet['E13'].value = self.Mapping_File_Name
        cover_sheet['E15'].value = self.Report_Creation_Time
       
        data_list = self.Source_Filename
        # Start writing from cell E11
        start_row = 17
        column = 'E'

        # Maximum cells to fill
        max_cells = 10
        if data_list:
            # Calculate the number of values per cell
            total_values = len(data_list)
            values_per_cell = total_values // max_cells
            extra_values = total_values % max_cells

            # Distribute values into cells
            index = 0
            for i in range(max_cells):
                num_values = values_per_cell + (1 if i < extra_values else 0)
                # Convert None to an empty string
                cell_values = ', '.join(str(value) if value is not None else '' for value in data_list[index:index + num_values])
                cell = f"{column}{start_row + i}"
                cover_sheet[cell].value = cell_values
                index += num_values
        else:
            # If the list is empty, do nothing or clear the cells if needed
            for i in range(max_cells):
                cell = f"{column}{start_row + i}"
                cover_sheet[cell].value = ''
            
        wb.save(excel_file)
        
        print(f'\n{strftime("%m/%d/%Y %I:%M:%S %p")} - Cover page updated \n')
        SP_SA_Audit_Log(self,f'Cover page updated ')
    
    except Exception as e:
        raise Exception(f"update_cover_page: {str(e)}")

def update_Consolidated_Report(self, excel_file, df, report_type, skip_rows=2):
    try:
        start_Time = datetime.now()

        # Load the Excel file
        wb = load_workbook(excel_file, keep_vba=True)

        # Select the active sheet (assuming it's the first sheet)
        sheet = wb['Consolidated']

        # Extract existing headers from the sheet after skipping specified rows
        headers = []
        for col in range(1, sheet.max_column + 1):
            header_cell = sheet.cell(row=skip_rows + 1, column=col)
            header_value = header_cell.value

            # Check if the cell above the header is part of a merged cell
            merged_value = None
            for merged_cell in sheet.merged_cells.ranges:
                # Debug: Print merged cell range

                if merged_cell.min_row == skip_rows and merged_cell.max_row == skip_rows:
                    if merged_cell.min_col <= col <= merged_cell.max_col:
                        merged_value = sheet.cell(row=skip_rows, column=merged_cell.min_col).value
                        # Debug: Print merged value
                        break

            # Prefix header with merged cell value if applicable
            if merged_value:
                #merged_value = re.sub(r'[ .()\-\s]', '_', merged_value)
                merged_value = re.sub(r'[ .()\-$+]', '_', merged_value)
                #merged_value = re.sub(r'/', '', merged_value)

                # Replace multiple underscores with a single underscore
                
                header_value = f"{merged_value}_{header_value}"
                header_value = re.sub(r'_+', '_', header_value)

            headers.append(header_value)

        # Debug: Print final headers
        # Match DataFrame columns to Excel columns
        common_columns = [col for col in headers if col in df.columns]
        # Get column indices in Excel file for matching columns
        col_indices = [headers.index(col) + 1 for col in common_columns]
        
        # Write data from DataFrame to Excel file
        for col_name, col_idx in zip(common_columns, col_indices):
            for row_idx, value in enumerate(df[col_name], start=skip_rows + 2):  # Start from skip_rows + 1 (headers row)
                #sheet.cell(row=row_idx, column=col_idx).value = value
                sheet.cell(row=row_idx, column=col_idx).value = value
        
        # Save the modified Excel file
        wb.save(excel_file)
       
        #db_TBL_Output_Report(self, report_type, os.path.basename(excel_file), 'S')
        SP_SA_Audit_Log(self, f'{report_type} Report updated')

def update_merged_cells(self,excel_file,sheet_name): 

        SP_SA_Audit_Log(self,f'Updating the merged cell values in {sheet_name}')
        print(f"\n{strftime("%m/%d/%Y %I:%M:%S %p")} - Updating the merged cell values in '{sheet_name}'\n")  
        
        # Load the workbook  
        workbook = load_workbook(excel_file, keep_vba=True)
        sheet = workbook[sheet_name]  # Ensure 'MEV' sheet exists  

        # List to hold the merged cell ranges in order  
        merged_ranges = list(sheet.merged_cells.ranges)  

        # Sort the merged ranges by the start row and then start column  
        merged_ranges.sort(key=lambda r: (r.bounds[1], r.bounds[0]))  # bounds gives min_row, min_col, max_row, max_col  

        occurrence = 1  # Initialize occurrence counter  

        # Update merged cells according to your specification  
        for merged_range in merged_ranges:  
            # Get the first cell in the merged range  
            first_cell = str(merged_range).split(':')[0]  
            original_value = sheet[first_cell].value  
            
            # Ensure original_value is not None before processing  
            if original_value is not None:  
                #print(f"Processing Merged Range: {merged_range}, First Cell: {first_cell}, Original Value: {original_value}")  

                # Check the conditions and replace values accordingly  
                if "YYYY" in original_value and "RIHC\\CNB" in original_value and occurrence == 1:  
                    if str(self.Reporting_Period)[-2:] == '12':
                        year = int(str(self.Reporting_Period)[:4]) + 1
                    else:
                        year = int(str(self.Reporting_Period)[:4])
                    # Replace for the first merged cell  
                    new_value = original_value.replace("YYYY", str(year)).replace("RIHC\\CNB", self.Data_Process)  
                    #print(f"Updating Merged Cell {occurrence}: New Value: {new_value}")  
                    sheet[first_cell].value = new_value  
                    occurrence += 1  # Increment occurrence for the next check  

                elif "YYYY" in original_value and "RIHC\\CNB" in original_value and occurrence == 2:  
                    if str(self.Reporting_Period)[-2:] == '12':
                        year = int(str(self.Reporting_Period)[:4])
                    else:
                        year = int(str(self.Reporting_Period)[:4]) - 1
                    # Replace for the second merged cell
                    new_value = original_value.replace("YYYY", str(year)).replace("RIHC\\CNB", self.Data_Process)
                    #print(f"Updating Merged Cell {occurrence}: New Value: {new_value}")
                    sheet[first_cell].value = new_value
                    occurrence += 1  # Increment occurrence for the next check

        # Save the workbook with the updated values  
        workbook.save(excel_file)  
        SP_SA_Audit_Log(self,f'Updated merged cell values in {sheet_name}')
        print(f"\n{strftime("%m/%d/%Y %I:%M:%S %p")} - Updated merged cell values in '{sheet_name}'\n")

                # Check the year in merged_value  
                if str(self.Current_year) in merged_value:  
                    merged_value = 'Current'  
                elif str(self.Previous_year) in merged_value:  
                    merged_value = 'Previous'  
                else:  
                    merged_value = 'Difference' 

                # Replace multiple underscores with a single underscore
                header_value = f"{header_value}_{merged_value}"
                header_value = re.sub(r'_+', '_', header_value)

            headers.append(header_value)
                
        # Match DataFrame columns to Excel columns
        common_columns = [col for col in headers if col in df.columns]
        # Get uncommon columns  
        uncommon_columns = list(set(headers) - set(common_columns)) + list(set(df.columns) - set(common_columns))  

        print('uncommon_columns: ',uncommon_columns)
        # Get column indices in Excel file for matching columns
        col_indices = [headers.index(col) + 1 for col in common_columns]
            
        # Define the fill patterns for yellow and orange  
        light_grey_fill = PatternFill(start_color='D3D3D3', end_color='D3D3D3', fill_type='solid')  # Light Grey

        previous_was_total = False  # Track if the last row was a "Total"  

        for col_name, col_idx in zip(common_columns, col_indices):  
            for row_idx, value in enumerate(df[col_name], start=skip_rows + 2):  # Start from skip_rows + 2 (headers row)  
                cell = sheet.cell(row=row_idx, column=col_idx)
                cell.value = value

                if col_name == "Scenario": 
                    # Check if the value contains the string "Total"  
                    if isinstance(value, str) and "Total" in value and value.endswith("Total"):  
                        if not previous_was_total:  
                            # Highlight the first "Total" row in yellow  
                            for col in range(1, sheet.max_column + 1):  # Highlighting all columns up to max column 
                                sheet.cell(row=row_idx, column=col).fill = light_grey_fill

                            # Highlight the second consecutive "Total" row in orange  
                            for col in range(1, sheet.max_column + 1):  # Highlighting all columns up to max column  
                                sheet.cell(row=row_idx, column=col).fill = light_blue_fill

                        previous_was_total = False  # Reset if current value is not "Total"    
                        
                if 'Assumption_grouping' in df.columns:        
                    if all(pd.isna(df.loc[row_idx - (skip_rows + 2), col]) for col in null_columns):
                        # Highlight the entire row in grey if all specified columns are NULL

def update_excel_with_df_data(self, excel_path, df_list,skip_rows=2):
        # Load your Excel file
        workbook = load_workbook(excel_path, keep_vba=True)
        Sensitivity_sheet = workbook['Sensitivity']
        agglist_df = pd.read_sql(f"""SELECT Aggregation_Group FROM DSKH_ARGUS_ETL_PROCESS_SA..VW_SA_{self.Data_Process}_Capital_Impact_aggregation GROUP BY Aggregation_Group ORDER BY REPLACE(Aggregation_Group,'Agg','')""",self.connectionSqlServer)
        printedrows = 0
        
        for idx, row in agglist_df.iterrows():
            agg_group = row['Aggregation_Group']
            df_filtr = df_list[df_list['Aggregation_Group'] == agg_group]
            if printedrows == 0:
                printrow = skip_rows + 2
            elif (printedrows != 0):
                printrow = printedrows
                
            if agg_group in ('Agg2.1', 'Agg3.1'):
                columns_to_replace = ['Sum.4', 'Sum.5', 'Sum.6', 'Sum.7']
                for col in columns_to_replace:
                    df_filtr.iloc[:, df_filtr.columns.get_loc(col)] = df_filtr.iloc[:, df_filtr.columns.get_loc(col)].replace(0, "")
            
            # Extract existing headers from the sheet after skipping specified rows
            headers = []
            for col in range(1, Sensitivity_sheet.max_column + 1):
                headers.append(Sensitivity_sheet.cell(row= skip_rows + 1, column=col).value)

            #print(headers)
            # Match DataFrame columns to Excel columns
            common_columns = [col for col in headers if col in df_filtr.columns]

            # Get column indices in Excel file for matching columns
            col_indices = [headers.index(col) + 1 for col in common_columns]
            
            # Define border styles
            thin_border = Border(left=Side(style='thin'), right=Side(style='thin'), top=Side(style='thin'), bottom=Side(style='thin'))

            # Write data from DataFrame to Excel file
            for col_name, col_idx in zip(common_columns, col_indices):
                for row_idx, value in enumerate(df_filtr[col_name], start = printrow):  # Start from skip_rows + 1 (headers row)
                    cell = Sensitivity_sheet.cell(row=row_idx, column=col_idx)
                    skp = 4
                    if agg_group == 'Agg3.1' and col_idx > skp and col_idx not in (14,25,36,47) :
                        if isinstance(value, (int, float)):
                            cell.value = f"{(value):.2f}%"
                        else:
                            cell.value = value
                    else:
                        cell.value = value
                    cell.border = thin_border
                    cell.fill = PatternFill(start_color='F2F2F2', end_color='F2F2F2', fill_type='solid')
                    cell.alignment = Alignment(wrap_text=True, horizontal='left', vertical='bottom')
            
            printedrows = printrow + df_filtr.shape[0] + 1

            if agg_group in ('Agg1.3', 'Agg2.1'):
                    # Merge the row from column A to AR and print the hardcoded value
                    Sensitivity_sheet.merge_cells(start_row=printedrows, start_column=1, end_row=printedrows, end_column=44)
                    cell = Sensitivity_sheet.cell(row=printedrows, column=1)
                    if agg_group == 'Agg1.3':
                        cell.value = "Cumul. bps Impact"
                    elif agg_group == 'Agg2.1':
                        cell.value = "Revised Capital Ratios"
                    cell.border = thin_border
                    cell.fill = PatternFill(start_color='99CCFF', end_color='99CCFF', fill_type='solid')
                    cell.alignment = Alignment(wrap_text=True, horizontal='left', vertical='center')
                    printedrows = printedrows + 1
    
        SP_SA_Audit_Log(self,'Output report is updated with two data frames')
        # Save the updated Excel file
        workbook.save(excel_path)

def datasheet_fill(self, excel_file, Sheet):     
        try:
            # Load the workbook and the first sheet
            wb = load_workbook(excel_file, keep_vba=True)
        
            # Duplicate the sheet 4 times with new names
            Sheet_names = Sheet 
        
            for names in Sheet_names:
                df1 = pd.read_sql(f"""select * from DSKH_ARGUS_ETL_PROCESS_SA..VW_SA_{self.Data_Process}_Capital_Impact_Chart where Aggregation_Scenario='{names}' and Aggregation_Group='Agg4.1'""",self.connectionSqlServer)
                df2 = pd.read_sql(f"""select * from DSKH_ARGUS_ETL_PROCESS_SA..VW_SA_{self.Data_Process}_Capital_Impact_Chart where Aggregation_Scenario='{names}' and Aggregation_Group='Agg4.2' ORDER BY Input_Type DESC""",self.connectionSqlServer)
                # Merge the two DataFrames
                merged_df = pd.concat([df1, df2], ignore_index=True)

                # Write only the matched columns to the corresponding sheet
                sheet = wb[names]
                headers = [cell.value for cell in sheet[1]]  # Assuming the first row contains headers
                matched_columns = [col for col in merged_df.columns if col in headers]

                # Get column indices in Excel file for matching columns
                col_indices = [headers.index(col) + 1 for col in matched_columns]

                for col_name, col_idx in zip(matched_columns, col_indices):
                    for row_idx, value in enumerate(merged_df[col_name], start=2):  # Start from row 2 to skip headers
                        sheet.cell(row=row_idx, column=col_idx, value=value)
                 
            # Save the new Excel file in the specified location
            wb.save(excel_file)
            
            print(f'\n{strftime("%m/%d/%Y %I:%M:%S %p")} - datasheet Filled \n')
            SP_SA_Audit_Log(self,f'datasheet Filled')
        
        except Exception as e:
            print(f"Error: {e}")

                raise Exception("The worksheet 'Pivot Chart' does not exist in the workbook.")
            
            Pivotchart_sheet = wb['Pivot Chart']
            
            # Define border styles
            thin_border = Border(left=Side(style='thin'), right=Side(style='thin'), top=Side(style='thin'), bottom=Side(style='thin'))

            # Loop through columns A to O and rows 4 to 100
            for col in range(1, 16):  # Columns A to O (1 to 15 in 1-based index)
                for row in range(4, 104):  # Rows 4 to 100
                    # Get the column letter
                    col_letter = chr(64 + col)  # Convert column number to letter (A=1, B=2, ...)
                    a = row - 2
                    # Write the formula into the cell
                    formula = f'=IFERROR(IF(INDIRECT("\'" & $A$2 & "\'!{col_letter}" & ROW({col_letter}{a})) = "", "", INDIRECT("\'" & $A$2 & "\'!{col_letter}" & ROW({col_letter}{a}))),"")'
                    Pivotchart_sheet[f'{col_letter}{row}'].value = formula
                    
                    # Ensure the formula is written correctly without '@'
                    Pivotchart_sheet[f'{col_letter}{row}'].value = Pivotchart_sheet[f'{col_letter}{row}'].value.replace('@', '')

                    # Apply border only if the cell contains data
                    if Pivotchart_sheet[f'{col_letter}{row}'].value not in (None, ""):
                        Pivotchart_sheet[f'{col_letter}{row}'].border = thin_border

                    # Apply custom format 0.00\% for columns E to N
                    if 'E' <= col_letter <= 'N':
                        Pivotchart_sheet[f'{col_letter}{row}'].number_format = '0.00\\%'

            #Hide O column in the sheet
            Pivotchart_sheet.column_dimensions['O'].hidden = True

            # Save the new Excel file in the specified location
            wb.save(excel_file)
            
            print(f'\n{strftime("%m/%d/%Y %I:%M:%S %p")} - Formula printed \n')
            SP_SA_Audit_Log(self,f'Formula printed')

        except Exception as e:
            raise Exception(f"Capital_Ratio_Report: {str(e)}")

def get_output_file_path(self):
    try:
        # Log the start of the template location fetching process
        print(f'\n{strftime("%m/%d/%Y %I:%M:%S %p")} - Fetching Template Location! \n')
        
        # Retrieve the template path from the application configuration
        template_path = get_app_value(self.AllWatcherLog, self.Report_Name, str(self.Report_Name) + '_Template_Location', 'Y')[0]
        print (template_path)
        # Check if the template file exists at the specified path and Raise an exception if the template file is not found
        if not (os.path.exists(template_path)):
            raise Exception('Template not found, Please check the Template path')
            
        SP_SA_Audit_Log(self,'Templates path Fetched')
            
        File_Location = get_app_value(self.AllWatcherLog,'SAOutputReport','SA_Output_Location','Y')[0]
            
        output_File_name = self.Filename
            
        output_File_path = os.path.join(File_Location,self.Parent_Report,self.Report_Name,self.Data_Process,str(self.Reporting_Period))
        
        try:
            if os.path.exists(output_File_path):
                print(f'\n{strftime("%m/%d/%Y %I:%M:%S %p")} - Directory exists: {output_File_path}\n')
                SP_SA_Audit_Log(self,f'Directory exists {output_File_path}')
            else:
                os.makedirs(output_File_path)
                print(f'\n{strftime("%m/%d/%Y %I:%M:%S %p")} - Directory {output_File_path} created\n')
                SP_SA_Audit_Log(self,f'Directory created {output_File_path}')
        except Exception as e:
            raise Exception(f"get_output_file_path: {str(e)}")
            
        output_File_path = os.path.join(output_File_path,output_File_name)

        with open(template_path, 'rb') as src, open(output_File_path, 'wb') as dest:
            dest.write(src.read())
            print(f'\n{strftime("%m/%d/%Y %I:%M:%S %p")} - Template copied successfully! \n')
        SP_SA_Audit_Log(self,f'Template copied successfully')
            
        return output_File_path

    except Exception as e:
        raise Exception(f"get_output_file_path: {str(e)}")
